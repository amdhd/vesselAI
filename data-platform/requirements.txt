# Use Python 3.10. dbt-core does not yet ship wheels for Python 3.14, so the
# project pins to 3.10 for the whole toolchain:
#   python3.10 -m venv .venv

# --- Warehouse engine (Phase 1: ingestion) ---
duckdb==1.5.4

# --- Transformations (Phase 2 silver, Phase 3 gold) via dbt ---
# dbt-duckdb pulls in a compatible dbt-core automatically; both are pinned here
# for reproducibility.
dbt-duckdb==1.10.1
dbt-core==1.12.0

# --- Transitive, pinned for a security fix ---
# urllib3 arrives via requests (dbt-core -> requests -> urllib3) and is not a
# direct dependency of anything here, but it ships in the warehouse-builder
# image and that image is scanned: 2.7.0 carries two HIGH advisories (proxy TLS
# interception, unbounded-memory DoS), both fixed in 2.8.0. requests allows
# urllib3<3, so the pin resolves cleanly. Drop it once the resolver picks a
# patched version on its own.
urllib3==2.8.0
